        // CART MANAGEMENT
        function addToCart(productId, customName = null) {
            let itemToAdd;
            if (customName) {
                itemToAdd = {
                    id: 'custom_' + Date.now(),
                    title: customName,
                    price: 249.90,
                    image: 'https://images.unsplash.com/photo-1589829085413-56de8ae18c73?auto=format&fit=crop&q=80&w=600',
                    qty: 1
                };
            } else {
                const prod = products.find(p => p.id === productId);
                if (!prod) return;
                itemToAdd = { ...prod, qty: 1 };
            }

            const existingIdx = cart.findIndex(c => c.id === itemToAdd.id);
            if (existingIdx > -1) {
                cart[existingIdx].qty += 1;
            } else {
                cart.push(itemToAdd);
            }

            saveData();
            updateCartBadge();
            showToast('Ürün sepete eklendi!');
        }

        function updateCartBadge() {
            const count = cart.reduce((acc, item) => acc + item.qty, 0);
            const badge = document.getElementById('cartBadge');
            if (count > 0) {
                badge.textContent = count;
                badge.classList.remove('hidden');
            } else {
                badge.classList.add('hidden');
            }
        }

        function toggleCartDrawer(show) {
            const drawer = document.getElementById('cartDrawer');
            if (show) {
                renderCartItems();
                drawer.classList.remove('hidden');
            } else {
                drawer.classList.add('hidden');
            }
        }

        function renderCartItems() {
            const container = document.getElementById('cartItemsList');
            const totalEl = document.getElementById('cartTotalPrice');

            if (cart.length === 0) {
                container.innerHTML = `
                    <div class="text-center py-12 text-gray-400 space-y-2">
                        <i class="fa-solid fa-cart-flatbed text-4xl"></i>
                        <p class="text-sm">Sepetiniz boş.</p>
                    </div>
                `;
                totalEl.textContent = '0.00 TL';
                return;
            }

            let total = 0;
            container.innerHTML = cart.map((item, index) => {
                const itemTotal = item.price * item.qty;
                total += itemTotal;
                return `
                    <div class="flex items-center space-x-3 bg-gray-50 dark:bg-slate-700/40 p-3 rounded-2xl border border-gray-200/60 dark:border-slate-700">
                        <img src="${item.image}" alt="" class="w-14 h-14 object-cover rounded-xl bg-gray-200">
                        <div class="flex-1 min-w-0">
                            <h4 class="text-xs font-bold text-gray-900 dark:text-white truncate">${item.title}</h4>
                            <p class="text-xs text-brand-600 dark:text-brand-400 font-semibold mt-0.5">${parseFloat(item.price).toFixed(2)} TL</p>
                            <div class="flex items-center space-x-2 mt-2">
                                <button onclick="updateQty(${index}, -1)" class="w-5 h-5 rounded-md bg-gray-200 dark:bg-slate-600 text-xs font-bold flex items-center justify-center">-</button>
                                <span class="text-xs font-semibold dark:text-white">${item.qty}</span>
                                <button onclick="updateQty(${index}, 1)" class="w-5 h-5 rounded-md bg-gray-200 dark:bg-slate-600 text-xs font-bold flex items-center justify-center">+</button>
                            </div>
                        </div>
                        <button onclick="removeFromCart(${index})" class="text-red-400 hover:text-red-600 p-1 text-sm">
                            <i class="fa-solid fa-trash"></i>
                        </button>
                    </div>
                `;
            }).join('');

            totalEl.textContent = `${total.toFixed(2)} TL`;
        }

        function updateQty(index, change) {
            cart[index].qty += change;
            if (cart[index].qty <= 0) cart.splice(index, 1);
            saveData();
            updateCartBadge();
            renderCartItems();
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            saveData();
            updateCartBadge();
            renderCartItems();
            showToast('Ürün sepetten çıkarıldı.');
        }

        function checkout() {
            if (cart.length === 0) {
                showToast('Sepetiniz boş!', 'fa-circle-xmark');
                return;
            }
            const total = cart.reduce((acc, i) => acc + (i.price * i.qty), 0);
            const newOrder = {
                id: 'SİP-' + Math.floor(100000 + Math.random() * 900000),
                date: new Date().toLocaleString('tr-TR'),
                items: [...cart],
                total: total
            };

            orders.unshift(newOrder);
            cart = [];
            saveData();
            updateCartBadge();
            toggleCartDrawer(false);
            renderOrders();
            showToast('Siparişiniz alındı! Teşekkür ederiz.', 'fa-circle-check');
        }

        function openEditProductModal(id) {
            const prod = products.find(p => p.id === id);
            if (!prod) return;

            document.getElementById('editProductId').value = prod.id;
            document.getElementById('prodTitle').value = prod.title;
            document.getElementById('prodPrice').value = prod.price;
            document.getElementById('prodCategory').value = prod.category;
            document.getElementById('prodDesc').value = prod.description;
            document.getElementById('prodImageUrl').value = prod.image;
            document.getElementById('productModalTitle').textContent = 'Ürün Düzenle';

            showImagePreview(prod.image);
            document.getElementById('productModal').classList.remove('hidden');
        }

        // IMAGE HANDLERS (URL & FileReader)
        function previewImageFromUrl() {
            const url = document.getElementById('prodImageUrl').value;
            if (url) showImagePreview(url);
            else hideImagePreview();
        }

        function handleFileSelect(e) {
            const file = e.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(evt) {
                    const dataUrl = evt.target.result;
                    document.getElementById('prodImageUrl').value = dataUrl;
                    showImagePreview(dataUrl);
                };
                reader.readAsDataURL(file);
            }
        }

        function showImagePreview(src) {
            const img = document.getElementById('imagePreview');
            const placeholder = document.getElementById('imagePreviewPlaceholder');
            img.src = src;
            img.classList.remove('hidden');
            placeholder.classList.add('hidden');
        }

        function hideImagePreview() {
            const img = document.getElementById('imagePreview');
            const placeholder = document.getElementById('imagePreviewPlaceholder');
            img.src = '';
            img.classList.add('hidden');
            placeholder.classList.remove('hidden');
        }

        function handleProductSubmit(e) {
            e.preventDefault();
            const editId = document.getElementById('editProductId').value;
            const title = document.getElementById('prodTitle').value;
            const price = parseFloat(document.getElementById('prodPrice').value);
            const category = document.getElementById('prodCategory').value;
            const description = document.getElementById('prodDesc').value;
            const image = document.getElementById('prodImageUrl').value || 'https://placehold.co/600x400?text=Resim+Yok';

            if (editId) {
                // EDIT EXISTING
                const idx = products.findIndex(p => p.id === editId);
                if (idx > -1) {
                    products[idx] = { id: editId, title, price, category, description, image };
                }
                showToast('Ürün başarıyla güncellendi.');
            } else {
                // ADD NEW
                const newProd = {
                    id: Date.now().toString(),
                    title,
                    price,
                    category,
                    description,
                    image
                };
                products.unshift(newProd);
                showToast('Yeni ürün eklendi.');
            }

            saveData();
            renderCategories();
            renderProducts();
            renderAdminProducts();
            closeProductModal();
        }

        function deleteProduct(id) {
            if (confirm('Bu ürünü silmek istediğinize emin misiniz?')) {
                products = products.filter(p => p.id !== id);
                saveData();
                renderCategories();
                renderProducts();
                renderAdminProducts();
                showToast('Ürün silindi.');
            }
        }
